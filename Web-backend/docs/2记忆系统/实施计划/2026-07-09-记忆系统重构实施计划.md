# 记忆系统重构 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `subagent-driven-development` (recommended) or inline task-by-task execution to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将当前内存 MVP 记忆模块重构为持久化、可检索、可治理、可审计，并真实接入 Agent 主链路的分层记忆系统。

**Architecture:** 以 `MemoryController` 作为唯一公开编排入口，底层拆分为 SQLite repository、候选抽取、时间治理、冲突合并、权限治理、语义索引、检索排序、上下文构造和审计报告。主链路必须通过 `MemoryController.build_memory_context(...)` 获取真实 `MemoryContext`，不允许创建空上下文冒充记忆系统已运行。

**Tech Stack:** Python dataclasses、标准库 `sqlite3`、现有 `services.embedding_provider`、现有 `core.contracts`、pytest。

## Global Constraints

- 不做补丁式修补；允许重构、删除和替换旧模块。
- 不保留旧 `InMemoryEventLog`、`InMemoryMemoryStore`、直接 `upsert_memory` 等冗余生产接口。
- 不引入未确认大型依赖；当前持久化使用 Python 标准库 SQLite。
- 记忆不能作为航空事实来源；事实仍只能来自 `evidence_package`。
- 主链路不得用空 `MemoryContext()` 冒充真实记忆上下文。
- 反馈产生的 `memory_update_candidate` 必须进入记忆治理流程。
- 配置缺失必须显式报错，不允许静默使用业务默认值。
- 每个任务完成后运行指定测试，并保持全量测试最终通过。

Create `src/memory/repository.py` with:

```python
from __future__ import annotations

import json
import sqlite3
from pathlib import Path
from typing import Any

from core.contracts import utc_now_iso
from memory.schemas import (
    InteractionEvent,
    MemoryRelation,
    SemanticMemoryEntry,
    StructuredMemory,
)


class SQLiteMemoryRepository:
    def __init__(self, path: str | Path) -> None:
        self.path = Path(path)

    def initialize(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self._connect() as conn:
            conn.executescript(
                """
                CREATE TABLE IF NOT EXISTS interaction_events (
                    event_id TEXT PRIMARY KEY,
                    payload_json TEXT NOT NULL,
                    created_at TEXT NOT NULL
                );
                CREATE TABLE IF NOT EXISTS structured_memories (
                    memory_id TEXT PRIMARY KEY,
                    user_id TEXT,
                    session_id TEXT,
                    memory_type TEXT NOT NULL,
                    status TEXT NOT NULL,
                    subject TEXT,
                    predicate TEXT,
                    object TEXT,
                    scope TEXT,
                    payload_json TEXT NOT NULL,
                    updated_at TEXT NOT NULL
                );
                CREATE TABLE IF NOT EXISTS memory_relations (
                    relation_id TEXT PRIMARY KEY,
                    from_id TEXT NOT NULL,
                    to_id TEXT NOT NULL,
                    relation_type TEXT NOT NULL,
                    payload_json TEXT NOT NULL
                );
                CREATE TABLE IF NOT EXISTS semantic_memory_entries (
                    semantic_id TEXT PRIMARY KEY,
                    memory_id TEXT NOT NULL,
                    text TEXT NOT NULL,
                    payload_json TEXT NOT NULL
                );
                CREATE TABLE IF NOT EXISTS memory_audit_logs (
                    audit_id INTEGER PRIMARY KEY AUTOINCREMENT,
                    action TEXT NOT NULL,
                    reason TEXT NOT NULL,
                    memory_id TEXT,
                    metadata_json TEXT NOT NULL,
                    created_at TEXT NOT NULL
                );
                """
            )

    def record_event(self, event: InteractionEvent) -> str:
        with self._connect() as conn:
            conn.execute(
                "INSERT OR REPLACE INTO interaction_events (event_id, payload_json, created_at) VALUES (?, ?, ?)",
                (event.event_id, json.dumps(event.to_dict(), ensure_ascii=False), utc_now_iso()),
            )
        return event.event_id

    def get_event(self, event_id: str) -> InteractionEvent:
        with self._connect() as conn:
            row = conn.execute(
                "SELECT payload_json FROM interaction_events WHERE event_id = ?",
                (event_id,),
            ).fetchone()
        if row is None:
            raise KeyError(event_id)
        return InteractionEvent.from_dict(json.loads(row["payload_json"]))  # type: ignore[return-value]

    def event_exists(self, event_id: str) -> bool:
        with self._connect() as conn:
            row = conn.execute("SELECT 1 FROM interaction_events WHERE event_id = ?", (event_id,)).fetchone()
        return row is not None

    def upsert_memory_record(self, memory: StructuredMemory) -> str:
        with self._connect() as conn:
            conn.execute(
                """
                INSERT OR REPLACE INTO structured_memories
                (memory_id, user_id, session_id, memory_type, status, subject, predicate, object, scope, payload_json, updated_at)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    memory.memory_id,
                    memory.user_id,
                    memory.session_id,
                    memory.memory_type.value,
                    memory.status.value,
                    memory.subject,
                    memory.predicate,
                    memory.object,
                    memory.scope,
                    json.dumps(memory.to_dict(), ensure_ascii=False),
                    utc_now_iso(),
                ),
            )
        return memory.memory_id

    def get_memory(self, memory_id: str) -> StructuredMemory:
        with self._connect() as conn:
            row = conn.execute(
                "SELECT payload_json FROM structured_memories WHERE memory_id = ?",
                (memory_id,),
            ).fetchone()
        if row is None:
            raise KeyError(memory_id)
        return StructuredMemory.from_dict(json.loads(row["payload_json"]))  # type: ignore[return-value]

    def list_memories(self, user_id: str | None = None) -> list[StructuredMemory]:
        sql = "SELECT payload_json FROM structured_memories"
        params: tuple[Any, ...] = ()
        if user_id is not None:
            sql += " WHERE user_id = ?"
            params = (user_id,)
        with self._connect() as conn:
            rows = conn.execute(sql, params).fetchall()
        return [StructuredMemory.from_dict(json.loads(row["payload_json"])) for row in rows]  # type: ignore[list-item]

    def insert_relation(self, relation: MemoryRelation) -> str:
        with self._connect() as conn:
            conn.execute(
                """
                INSERT OR REPLACE INTO memory_relations
                (relation_id, from_id, to_id, relation_type, payload_json)
                VALUES (?, ?, ?, ?, ?)
                """,
                (
                    relation.relation_id,
                    relation.from_id,
                    relation.to_id,
                    relation.relation_type,
                    json.dumps(relation.to_dict(), ensure_ascii=False),
                ),
            )
        return relation.relation_id

    def insert_semantic_entry(self, entry: SemanticMemoryEntry) -> str:
        with self._connect() as conn:
            conn.execute(
                """
                INSERT OR REPLACE INTO semantic_memory_entries
                (semantic_id, memory_id, text, payload_json)
                VALUES (?, ?, ?, ?)
                """,
                (entry.semantic_id, entry.memory_id, entry.text, json.dumps(entry.to_dict(), ensure_ascii=False)),
            )
        return entry.semantic_id

    def list_semantic_entries(self) -> list[SemanticMemoryEntry]:
        with self._connect() as conn:
            rows = conn.execute("SELECT payload_json FROM semantic_memory_entries").fetchall()
        return [SemanticMemoryEntry.from_dict(json.loads(row["payload_json"])) for row in rows]  # type: ignore[list-item]

    def record_audit(self, action: str, reason: str, memory_id: str | None = None, metadata: dict[str, Any] | None = None) -> None:
        with self._connect() as conn:
            conn.execute(
                """
                INSERT INTO memory_audit_logs (action, reason, memory_id, metadata_json, created_at)
                VALUES (?, ?, ?, ?, ?)
                """,
                (action, reason, memory_id, json.dumps(metadata or {}, ensure_ascii=False), utc_now_iso()),
            )

    def list_audit_logs(self) -> list[dict[str, Any]]:
        with self._connect() as conn:
            rows = conn.execute(
                "SELECT action, reason, memory_id, metadata_json, created_at FROM memory_audit_logs ORDER BY audit_id"
            ).fetchall()
        return [
            {
                "action": row["action"],
                "reason": row["reason"],
                "memory_id": row["memory_id"],
                "metadata": json.loads(row["metadata_json"]),
                "created_at": row["created_at"],
            }
            for row in rows
        ]

    def _connect(self) -> sqlite3.Connection:
        conn = sqlite3.connect(self.path)
        conn.row_factory = sqlite3.Row
        return conn
```

from __future__ import annotations

from memory.schemas import InteractionEvent, MemoryCandidate, MemoryType


class MemoryExtractor:
    def extract(self, event: InteractionEvent) -> list[MemoryCandidate]:
        payload = event.payload
        memory_type = self._memory_type(event)
        key = payload.get("key", event.event_type)
        value = payload.get("value", payload.get("text", ""))
        subject = f"user:{event.user_id}" if event.user_id else "user:anonymous"
        return [
            MemoryCandidate(
                source_event_ids=[event.event_id],
                source_event_id=event.event_id,
                memory_type=memory_type,
                key=key,
                subject=subject,
                predicate=key,
                object=str(value),
                scope=payload.get("scope", "session" if event.session_id else "global"),
                value=value,
                observed_at=event.observed_at,
                confidence=float(payload.get("confidence", event.confidence)),
                stability_score=float(payload.get("stability_score", 0.5)),
                privacy_level=event.privacy_level,
                user_id=event.user_id,
                session_id=event.session_id,
                write_agent=payload.get("write_agent", "memory_extractor"),
                evidence_count=int(payload.get("evidence_count", 1)),
            )
        ]

    def _memory_type(self, event: InteractionEvent) -> MemoryType:
        payload = event.payload
        if payload.get("fact_candidate"):
            return MemoryType.AVIATION_FACT
        if "memory_type" in payload:
            return MemoryType(payload["memory_type"])
        if event.event_type in {"preference", "preference_signal"}:
            return MemoryType.PREFERENCE
        if event.event_type in {"misconception", "weak_point"}:
            return MemoryType.MISCONCEPTION
        if event.event_type in {"policy", "authorization"}:
            return MemoryType.POLICY
        return MemoryType.SUMMARY

from __future__ import annotations

from dataclasses import dataclass

from memory.schemas import MemoryCandidate, MemoryType, PrivacyLevel


@dataclass
class GovernanceDecision:
    accepted: bool
    reason: str


class MemoryGovernanceChecker:
    def __init__(self, low_confidence_threshold: float = 0.6) -> None:
        self.low_confidence_threshold = low_confidence_threshold
        self.allowed_long_term_types = {
            MemoryType.PREFERENCE,
            MemoryType.MISCONCEPTION,
            MemoryType.SUMMARY,
            MemoryType.POLICY,
            MemoryType.REFLECTION,
            MemoryType.PROMPT_ASSET_CANDIDATE,
        }

    def check(self, candidate: MemoryCandidate) -> GovernanceDecision:
        if candidate.memory_type not in self.allowed_long_term_types:
            return GovernanceDecision(False, "memory_type_not_allowed")
        if candidate.confidence < self.low_confidence_threshold:
            return GovernanceDecision(False, "low_confidence")
        if candidate.privacy_level == PrivacyLevel.HIGH:
            return GovernanceDecision(False, "high_privacy_requires_authorization")
        return GovernanceDecision(True, "accepted")
