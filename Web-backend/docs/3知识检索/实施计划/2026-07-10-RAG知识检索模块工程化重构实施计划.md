# RAG 知识检索模块工程化重构 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `subagent-driven-development` only after the user explicitly authorizes subagents; otherwise execute inline task-by-task. Every task uses checkbox (`- [ ]`) tracking and must pass its own review gate before the next task begins.

**Goal:** 将当前内存型 Mock 文本 RAG 升级为配置驱动、SQLite 持久化、支持 BM25/稠密向量/父子节点/场景/轻量图谱/视觉页面的可审计混合检索系统，并实现 RRF、四态证据门控和定向补检索。

**Architecture:** 以 `RetrievalController` 作为业务侧唯一检索入口，内部拆分 typed config、持久化 repository、query rewriter、adaptive planner、channel executor、RRF fusion、feature reranker、EvidencePackageBuilder 和 EvidenceGate。默认离线实现使用标准库 SQLite 与 FTS5，不引入未批准的大型服务；真实 embedding、OCR/layout 和外部向量库均通过现有 Provider/adapter 边界接入。

**Tech Stack:** Python 3.13、dataclasses、标准库 `sqlite3`、SQLite FTS5、现有 `EmbeddingProvider`/`VectorStore`、pytest、现有 Agent 状态机与 trace。

```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Any

from core.errors import ConfigError
from core.settings import parse_simple_yaml


@dataclass(frozen=True)
class RepositoryConfig:
    provider: str
    path: Path


@dataclass(frozen=True)
class ChunkingConfig:
    min_chars: int
    target_chars: int
    max_chars: int
    overlap_chars: int


@dataclass(frozen=True)
class ChannelConfig:
    enabled: bool
    top_k: int
    max_hops: int = 0


@dataclass(frozen=True)
class RetrievalConfig:
    rrf_k: int
    fusion_top_k: int
    rerank_top_k: int
    max_retrieve_loops: int
    channels: dict[str, ChannelConfig]


@dataclass(frozen=True)
class EmbeddingConfig:
    provider: str
    dimension: int


@dataclass(frozen=True)
class GateConfig:
    allowed_core_review_status: tuple[str, ...]
    min_core_items: int
    require_parameter_source: bool
    require_visual_text_cross_check: bool


@dataclass(frozen=True)
class RerankConfig:
    weights: dict[str, float]


@dataclass(frozen=True)
class RagConfig:
    repository: RepositoryConfig
    chunking: ChunkingConfig
    retrieval: RetrievalConfig
    embedding: EmbeddingConfig
    gate: GateConfig
    rerank: RerankConfig


def _required(mapping: dict[str, Any], key: str, scope: str) -> Any:
    if key not in mapping:
        raise ConfigError(f"{scope}.{key}", "required key is missing")
    return mapping[key]


def load_rag_config(
    path: str | Path = "configs/rag.yaml",
    profile: str = "offline",
) -> RagConfig:
    config_path = Path(path)
    data = parse_simple_yaml(config_path.read_text(encoding="utf-8"))
    rag = _required(data, "rag", "config")
    repository = _required(rag, "repository", "rag")
    chunking = _required(rag, "chunking", "rag")
    retrieval = _required(rag, "retrieval", "rag")
    embedding = _required(rag, "embedding", "rag")
    gate = _required(rag, "gate", "rag")
    rerank = _required(rag, "rerank", "rag")

    channels = {
        name: ChannelConfig(
            enabled=bool(values["enabled"]),
            top_k=int(values["top_k"]),
            max_hops=int(values.get("max_hops", 0)),
        )
        for name, values in retrieval["channels"].items()
    }
    weights = {name: float(value) for name, value in rerank["weights"].items()}
    if abs(sum(weights.values()) - 1.0) > 1e-9:
        raise ConfigError("rag.rerank.weights", "weights must sum to 1.0")
    if any(channel.top_k <= 0 for channel in channels.values()):
        raise ConfigError("rag.retrieval.channels", "top_k must be positive")
    if profile == "production" and embedding["provider"] == "mock":
        raise ConfigError("rag.embedding.provider", "mock embedding is forbidden in production")

    return RagConfig(
        repository=RepositoryConfig(
            provider=str(repository["provider"]),
            path=Path(str(repository["path"])),
        ),
        chunking=ChunkingConfig(**{key: int(value) for key, value in chunking.items()}),
        retrieval=RetrievalConfig(
            rrf_k=int(retrieval["rrf_k"]),
            fusion_top_k=int(retrieval["fusion_top_k"]),
            rerank_top_k=int(retrieval["rerank_top_k"]),
            max_retrieve_loops=int(retrieval["max_retrieve_loops"]),
            channels=channels,
        ),
        embedding=EmbeddingConfig(
            provider=str(embedding["provider"]),
            dimension=int(embedding["dimension"]),
        ),
        gate=GateConfig(
            allowed_core_review_status=tuple(gate["allowed_core_review_status"]),
            min_core_items=int(gate["min_core_items"]),
            require_parameter_source=bool(gate["require_parameter_source"]),
            require_visual_text_cross_check=bool(gate["require_visual_text_cross_check"]),
        ),
        rerank=RerankConfig(weights=weights),
    )
```

```python
class KnowledgeRepository:
    def __init__(self, path: Path) -> None:
        self.path = path
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.connection = sqlite3.connect(path)
        self.connection.row_factory = sqlite3.Row
        self._create_schema()

    def _create_schema(self) -> None:
        with self.connection:
            self.connection.executescript(SCHEMA_SQL)

    def register_source(self, source: SourceRecord) -> str:
        with self.connection:
            self.connection.execute(
                """
                INSERT INTO knowledge_sources (
                    source_id, source_type, title, authority_level,
                    review_status, version, created_at, updated_at,
                    content_hash, original_uri, metadata_json
                ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                ON CONFLICT(source_id) DO UPDATE SET
                    title=excluded.title,
                    authority_level=excluded.authority_level,
                    review_status=excluded.review_status,
                    version=excluded.version,
                    updated_at=excluded.updated_at,
                    metadata_json=excluded.metadata_json
                """,
                (
                    source.source_id,
                    source.source_type,
                    source.title,
                    source.authority_level,
                    source.review_status.value,
                    source.version,
                    source.created_at,
                    source.updated_at,
                    source.content_hash,
                    source.original_uri,
                    json.dumps(source.metadata, ensure_ascii=False, sort_keys=True),
                ),
            )
        return source.source_id

    def is_core_source(self, source_id: str) -> bool:
        source = self.get_source(source_id)
        return bool(
            source
            and source.review_status == ReviewStatus.REVIEWED
            and source.source_type not in {"user_feedback", "model_generated", "memory"}
        )

    def close(self) -> None:
        self.connection.close()
```

Implement:

```python
HEADING_PATTERN = re.compile(r"^\s{0,3}(#{1,6})\s+(.+?)\s*$")
SENTENCE_BOUNDARY = re.compile(r"(?<=[。！？；])")


def split_sections(document: str) -> list[tuple[str, str]]:
    sections: list[tuple[str, list[str]]] = []
    current_title = "正文"
    current_lines: list[str] = []
    for line in document.splitlines():
        match = HEADING_PATTERN.match(line)
        if match:
            if current_lines:
                sections.append((current_title, current_lines))
            current_title = match.group(2).strip()
            current_lines = []
        elif line.strip():
            current_lines.append(line.strip())
    if current_lines:
        sections.append((current_title, current_lines))
    return [(title, "\n".join(lines)) for title, lines in sections]
```

Implement deterministic packing that never combines content across two section tuples and only applies overlap inside the same parent:

```python
def pack_sentences(text: str, config: ChunkingConfig) -> list[str]:
    sentences = [item.strip() for item in SENTENCE_BOUNDARY.split(text) if item.strip()]
    chunks: list[str] = []
    current = ""
    for sentence in sentences:
        if len(sentence) > config.max_chars:
            if current:
                chunks.append(current)
                current = ""
            step = max(1, config.max_chars - config.overlap_chars)
            chunks.extend(
                sentence[start : start + config.max_chars]
                for start in range(0, len(sentence), step)
                if sentence[start : start + config.max_chars]
            )
            continue
        proposed = sentence if not current else current + sentence
        if current and len(proposed) > config.target_chars:
            chunks.append(current)
            overlap = current[-config.overlap_chars :] if config.overlap_chars else ""

    if current:
        chunks.append(current)

```

```python
def reciprocal_rank_fusion(
    channel_hits: dict[str, list[RetrievalHit]],
    rrf_k: int,
) -> list[FusedHit]:
    merged: dict[str, FusedHit] = {}
    for channel, hits in channel_hits.items():
        for rank, hit in enumerate(hits, start=1):
            item = merged.setdefault(
                hit.item_id,
                FusedHit(item_id=hit.item_id, source_id=hit.source_id),
            )
            item.rrf_score += 1.0 / (rrf_k + rank)
            item.channels.append(channel)
            item.channel_ranks[channel] = rank
            item.hits.append(hit)
    return sorted(merged.values(), key=lambda item: (-item.rrf_score, item.item_id))
```

```python
from pypdf import PdfReader


class PdfIngestor:
    def parse_file(self, path: Path, source_id: str) -> list[ParsedPdfPage]:
        reader = PdfReader(path)
        pages: list[ParsedPdfPage] = []
        for index, page in enumerate(reader.pages, start=1):
            text = page.extract_text() or ""
            incomplete = [] if text.strip() else ["missing_extractable_text", "ocr_required"]
            pages.append(
                ParsedPdfPage(
                    source_id=source_id,
                    page_number=index,
                    ocr_text=text,
                    layout_trace=LayoutTrace(
                        source_id=source_id,
                        page_id=f"{source_id}:page:{index}",
                        parser="pypdf",
                        ocr_enabled=False,
                        layout_enabled=False,
                        incomplete_reasons=incomplete,
                    ),
                )
            )
        return pages
```
